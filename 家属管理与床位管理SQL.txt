-- 家属管理与床位管理初始化脚本
-- MySQL 8.0+。请在执行前备份 elder 数据库。

CREATE DATABASE IF NOT EXISTS `elder` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE `elder`;

CREATE TABLE IF NOT EXISTS `family_member` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键',
  `elder_id` bigint NOT NULL COMMENT '关联老人ID',
  `name` varchar(50) NOT NULL COMMENT '家属姓名',
  `relation` varchar(30) NOT NULL COMMENT '与老人关系',
  `phone` varchar(20) NOT NULL COMMENT '联系电话',
  `id_card_no` varchar(32) DEFAULT NULL COMMENT '身份证号',
  `is_primary` tinyint NOT NULL DEFAULT 0 COMMENT '是否主要联系人：0否，1是',
  `address` varchar(255) DEFAULT NULL COMMENT '家庭住址',
  `remark` varchar(500) DEFAULT NULL COMMENT '备注',
  `deleted` tinyint NOT NULL DEFAULT 0 COMMENT '逻辑删除：0否，1是',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_family_member_elder_id` (`elder_id`),
  KEY `idx_family_member_phone` (`phone`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='老人家属表';

-- elder_id 唯一，保证一位老人同一时间只能占用一个床位。
CREATE TABLE IF NOT EXISTS `bed` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键',
  `building` varchar(50) NOT NULL COMMENT '楼栋',
  `floor` int NOT NULL COMMENT '楼层',
  `room_no` varchar(30) NOT NULL COMMENT '房间号',
  `bed_no` varchar(30) NOT NULL COMMENT '床位号',
  `bed_type` varchar(50) DEFAULT NULL COMMENT '床位类型',
  `monthly_price` decimal(10,2) DEFAULT NULL COMMENT '月费',
  `status` tinyint NOT NULL DEFAULT 0 COMMENT '0空闲，1已入住，2停用',
  `elder_id` bigint DEFAULT NULL COMMENT '当前入住老人ID',
  `remark` varchar(500) DEFAULT NULL COMMENT '备注',
  `deleted` tinyint NOT NULL DEFAULT 0 COMMENT '逻辑删除：0否，1是',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_bed_location` (`building`, `floor`, `room_no`, `bed_no`),
  UNIQUE KEY `uk_bed_elder_id` (`elder_id`),
  KEY `idx_bed_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='床位表';

-- 为既有老人表增加当前床位字段。若该列已经存在，请跳过以下两条语句。
ALTER TABLE `elder` ADD COLUMN `bed_id` bigint DEFAULT NULL COMMENT '当前入住床位ID' AFTER `remark`;

INSERT INTO `role_permission` (`role_id`, `permission_id`, `create_time`, `update_time`)
SELECT r.`id`, p.`id`, NOW(), NOW()
FROM `role` r CROSS JOIN `permission` p
WHERE r.`deleted` = 0 AND p.`deleted` = 0
  AND (p.`path` IN ('/family-member', '/bed') OR p.`permission_value` IN ('familyMember:add', 'familyMember:update', 'familyMember:delete', 'bed:add', 'bed:update', 'bed:assign', 'bed:delete'))
  AND NOT EXISTS (SELECT 1 FROM `role_permission` rp WHERE rp.`role_id` = r.`id` AND rp.`permission_id` = p.`id`);
